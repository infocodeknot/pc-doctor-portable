💻 Source Preview — Main.ps1 (Entry Point)

# PC Doctor Portable — Main Entry Point
# Version 1.2 | Code Knot Technology

param(
    [switch]$AutoMode,
    [switch]$InteractiveMode,
    [switch]$SelfTest,
    [switch]$RenderShot,
    [string]$ShotPath = "RenderShot.png"
)

# Load configuration
$scriptDir = Split-Path -Parent $MyInvocation.MyCommand.Path
$configPath = Join-Path $scriptDir "Config.json"

if (Test-Path $configPath) {
    $script:Config = Get-Content $configPath -Raw | ConvertFrom-Json
    Write-Host "[OK] Configuration loaded" -ForegroundColor Green
} else {
    Write-Host "[WARN] Config.json not found" -ForegroundColor Yellow
}

# Import all modules
$moduleDir = Join-Path $scriptDir "Modules"
Get-ChildItem -Path $moduleDir -Filter "*.ps1" | ForEach-Object {
    . $_.FullName
}

# Launch GUI or Console based on mode
if ($SelfTest) {
    # Run built-in GUI self-test
    & (Join-Path $scriptDir "App\PCDoctor-GUI.ps1") -SelfTest
} elseif (-not $AutoMode -and -not $InteractiveMode) {
    # Default: show menu
    Show-Menu
} else {
    Invoke-AllModules -AutoMode:$AutoMode
}

🚀 Quick Start Commands

# Run GUI
powershell -ExecutionPolicy Bypass -File "App\PCDoctor-GUI.ps1"

# Run Console
.\Run.bat

# Run all 47 Pester tests
pwsh -File Tools\Run-Tests.ps1

# Run GUI self-test (no window)
powershell -File App\PCDoctor-GUI.ps1 -SelfTest

# Build installer (requires Inno Setup)
pwsh -File Tools\Build-Installer.ps1
